cmake_minimum_required(VERSION 3.16)

project(open_draft_cpp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(OPEN_DRAFT_CPP_LIB_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../out/cpp/lib")
set(OPEN_DRAFT_CPP_BIN_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../out/cpp/bin")
file(MAKE_DIRECTORY "${OPEN_DRAFT_CPP_LIB_OUTPUT_DIR}")
file(MAKE_DIRECTORY "${OPEN_DRAFT_CPP_BIN_OUTPUT_DIR}")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${OPEN_DRAFT_CPP_LIB_OUTPUT_DIR}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${OPEN_DRAFT_CPP_BIN_OUTPUT_DIR}")
foreach(CONFIG_TYPE ${CMAKE_CONFIGURATION_TYPES})
	string(TOUPPER "${CONFIG_TYPE}" CONFIG_UPPER)
	set("CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${CONFIG_UPPER}" "${OPEN_DRAFT_CPP_LIB_OUTPUT_DIR}")
	set("CMAKE_RUNTIME_OUTPUT_DIRECTORY_${CONFIG_UPPER}" "${OPEN_DRAFT_CPP_BIN_OUTPUT_DIR}")
endforeach()

add_subdirectory(src)
add_subdirectory(test)

string(TIMESTAMP RUN_ID "%Y-%m-%d_%H-%M-%S")
set(TEST_LOG_ROOT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../out/logs/cpp/${RUN_ID}")
file(MAKE_DIRECTORY "${TEST_LOG_ROOT_DIR}")
add_custom_target(unit-tests
	COMMAND $<TARGET_FILE:core_ut> --report-file "${TEST_LOG_ROOT_DIR}/TEST-core_ut_report.xml"
	DEPENDS core_ut
	WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
	COMMENT "Running C++ core unit tests"
	COMMAND $<TARGET_FILE:lang_ut> --report-file "${TEST_LOG_ROOT_DIR}/TEST-lang_ut_report.xml"
	DEPENDS lang_ut
	WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}
	COMMENT "Running C++ language unit tests"
)
